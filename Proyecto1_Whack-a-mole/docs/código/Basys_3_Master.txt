###############################################################################
# CLOCK - 100 MHz
###############################################################################

set_property -dict { PACKAGE_PIN W5 IOSTANDARD LVCMOS33 } [get_ports clk]

create_clock -add -name sys_clk_pin \
    -period 10.00 \
    -waveform {0 5} \
    [get_ports clk]


###############################################################################
# RESET
# Boton central BTNC de la Basys 3
###############################################################################

set_property -dict { PACKAGE_PIN U18 IOSTANDARD LVCMOS33 } [get_ports reset_btn]


###############################################################################
# LED DE ESTADO
# LED0
# game_status_led = 1 -> GAME OVER
###############################################################################

set_property -dict { PACKAGE_PIN U16 IOSTANDARD LVCMOS33 } [get_ports game_status_led]

set_property -dict { PACKAGE_PIN U5 IOSTANDARD LVCMOS33 } [get_ports {seg[4]}]

set_property -dict { PACKAGE_PIN U7 IOSTANDARD LVCMOS33 } [get_ports {seg[6]}]

# PMOD JB
# Comunicacion con el circuito discreto
###############################################################################

# JB1
# TX del circuito discreto -> RX de la FPGA
set_property -dict { PACKAGE_PIN A14 IOSTANDARD LVCMOS33 } [get_ports serial_data]

# JB2
# FPGA -> circuito discreto
# Solicitud de una nueva posicion
